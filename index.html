async function load() { const btn = $("refresh"); try { btn.disabled = true; btn.textContent = "Memuat data..."; $("error").style.display = "none"; const url = GOOGLE_SHEETS_API_URL + "?t=" + Date.now(); const response = await fetch(url, { method: "GET", cache: "no-store", redirect: "follow" }); if (!response.ok) { throw new Error("HTTP " + response.status); } const result = await response.json(); if (!result || result.success !== true) { throw new Error( result?.error || "Respons API tidak memiliki success: true." ); } if (!Array.isArray(result.data)) { throw new Error( "Properti data bukan array. Periksa format respons Apps Script." ); } const records = result.data; const normalized = records.map(norm); if (records.length > 0 && normalized.every( r => !r.id && !r.sev && !r.region )) { throw new Error( "Data diterima, tetapi header SQDT, SEVERITY, dan REGION tidak cocok." ); } all = normalized.filter( r => r.id || r.sev || r.region || r.branch || r.kab ); if (!all.length) { throw new Error( "API berhasil, tetapi tidak ada baris yang dikenali dashboard." ); } fill("severityFilter", uniq(all.map(r => r.sev))); fill("regionFilter", uniq(all.map(r => r.region))); fill("statusFilter", uniq(all.map(r => r.status))); fill("circleFilter", uniq(all.map(r => r.circle))); fill("rcaFilter", uniq(all.map(r => r.rca))); fill("branchFilter", uniq(all.map(r => r.branch))); render(); const stamp = result.updatedAt ? new Date(result.updatedAt).toLocaleString("id-ID") : new Date().toLocaleString("id-ID"); $("rowCount").textContent = `Menampilkan ${all.length.toLocaleString("id-ID")} baris · diperbarui ${stamp}`; console.log("Google Sheets berhasil dimuat."); console.log("Total baris sumber:", records.length); console.log("Total baris dashboard:", all.length); console.log("Header data:", Object.keys(records[0] || {})); console.log("Contoh data:", records[0] || {}); } catch (error) { console.error("Gagal memuat Google Sheets:", error); showError( "Gagal memuat data: " + error.message + ". Periksa deployment Apps Script, format API, dan izin akses." ); } finally { btn.disabled = false; btn.textContent = "↻ Refresh data"; } }
